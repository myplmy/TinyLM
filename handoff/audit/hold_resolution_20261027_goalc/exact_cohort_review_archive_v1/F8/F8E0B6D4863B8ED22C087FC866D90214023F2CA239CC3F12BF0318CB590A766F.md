- 전체 평탄화 코드: import urllib2 import json import gzip import StringIO def unzipper(url="http://example.com/python_list_turned_into.json.gz"): request = urllib2.Request(url) request.add_header('Accept-encoding', 'gzip') opener = urllib2.build_opener() f = opener.open(request) compresseddata = f.read() compressedstream = StringIO.StringIO(compresseddata) gzipper = gzip.GzipFile(fileobj=compressedstream) data = gzipper.read() list_output = json.loads(data) return list_output
