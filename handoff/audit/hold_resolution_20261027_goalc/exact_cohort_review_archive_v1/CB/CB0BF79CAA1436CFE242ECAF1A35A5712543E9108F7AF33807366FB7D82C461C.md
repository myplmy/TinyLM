## 손상된 demo 데이터 준비 코드

> library(dplyr) <- c("Susan", "Susan", "Madison", "Madison", demo_names "Hillary", "Hillary", "Hillary") <- c(rep(c(1930, 2000), 3), 1930) demo_years <- tibble(first_names = demo_names, demo_df last_names = LETTERS[1:7], years = demo_years, min_years = demo_years - 3, max_years = demo_years + 3) demo_df

## 시작 입력/구분이 빠진 두 dplyr pipeline

> %>% demo_df distinct(first_names, years) %>% rowwise() %>% do(results = gender(.$first_names, years = .$years, method = "demo")) %>% do(bind_rows(.$results))

> %>% demo_df distinct(first_names, years) %>% group_by(years) %>% do(results = gender(.$first_names, years = .$years, method = "demo")) %>% do(bind_rows(.$results))
