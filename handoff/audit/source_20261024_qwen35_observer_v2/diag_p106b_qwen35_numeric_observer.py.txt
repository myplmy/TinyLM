def identity():
    if importlib.metadata.version("torch") != TORCH_VERSION:
        raise q.Hold("documented private API diagnosis requires the pinned Torch version")
    base = q.identity()
    if base["sha256"] != BASE_SOURCE:
        raise ValueError("frozen Core6 v3 source changed")
    files = dict(base["files"])
    files.update({name: q.digest(ROOT / name) for name in OWN})
    capture = Path(importlib.metadata.distribution("transformers").locate_file("transformers/utils/output_capturing.py"))
    files[str(capture)] = q.digest(capture)
    dist = importlib.metadata.distribution("torch")
    files.update({str(dist.locate_file("torch/" + name)): q.digest(dist.locate_file("torch/" + name)) for name in TORCH_FILES})
    result = dict(base_core6_source_sha256=BASE_SOURCE, files=files, versions=base["versions"])
    return dict(result, sha256=q.object_sha(result))


def error_snapshot(error):
    """Preserve error text, then release completed traceback frames and chains."""
    text = "".join(traceback.format_exception(type(error), error, error.__traceback__))
    result = dict(type=type(error).__name__, message=str(error)[:2000], traceback=text[:65536],
                  traceback_text_truncated=len(text) > 65536)
    pending, seen = [error], set()
    while pending:
        current = pending.pop()
        if id(current) in seen:
            continue
        seen.add(id(current))
        pending.extend(value for value in (current.__cause__, current.__context__) if value is not None)
        traceback.clear_frames(current.__traceback__)
        current.__traceback__ = current.__cause__ = current.__context__ = None
    return result

def number(value):
    value = float(value)
    return value if math.isfinite(value) else None


def cleanup_api(torch):
    if str(torch.__version__) != TORCH_VERSION:
        raise q.Hold("actual imported Torch version differs; private clear not called")
    clear = getattr(torch._C, "_cuda_clearCublasWorkspaces", None)
    if not callable(clear):
        raise q.Hold("documented private cuBLAS clear API absent; no replacement")
    return clear


def imported_torch_identity(torch, signature):
    receipts = {}
    for module in (torch, torch.version, torch.cuda, torch.cuda.memory):
        path = str(Path(module.__file__).absolute())
        if path not in signature["files"] or q.digest(path) != signature["files"][path]:
            raise ValueError("actual imported Torch source/path differs: " + path)
        receipts[path] = signature["files"][path]
    return receipts


def cuda_storage_metadata(torch, *, objects_fn=gc.get_objects, maximum_objects=500000, maximum_storages=128):
    objects = value = storage = None
    rows, seen, tensors, scanned = [], set(), 0, 0
    complete = True
    try:
        objects = objects_fn()
        for value in objects:
            scanned += 1
            if scanned > maximum_objects:
                complete = False
                break
            if not isinstance(value, torch.Tensor) or not value.is_cuda:
                continue
            tensors += 1
            storage = value.untyped_storage()
            key = (str(value.device), int(storage.data_ptr()), int(storage.nbytes()))
            if key in seen:
                storage = None
                continue
            if len(rows) == maximum_storages:
                complete = False
                break
            seen.add(key)
            rows.append(dict(device=key[0], storage_address=key[1], storage_bytes=key[2],
                             shape=list(value.shape[:16]), rank=len(value.shape), dtype=str(value.dtype)))
            storage = None
        return dict(scope="GC_TRACKED_PYTHON_TENSORS_ONLY_NOT_ALL_CPP_ALLOCATIONS", complete=complete,
                    scanned_objects=min(scanned, maximum_objects), cuda_tensor_objects=tensors,
                    unique_storages=len(rows), unique_storage_bytes=sum(row["storage_bytes"] for row in rows), storages=rows,
                    metadata_limits=dict(objects=maximum_objects, storages=maximum_storages, shape_dimensions=16))
    finally:
        storage = value = None
        if objects is not None:
            objects.clear()
        objects = None


def registered_devices(model):
    result, value, items = {}, None, None
    try:
        for kind in ("parameters", "buffers"):
            counts, samples = {}, []
            items = model.named_parameters() if kind == "parameters" else model.named_buffers()
            for name, value in items:
                device = str(value.device)
                counts[device] = counts.get(device, 0) + 1
                if value.device.type != "cpu" and len(samples) < 16:
                    samples.append(dict(name=name, device=device, shape=list(value.shape[:16])))
            result[kind] = dict(device_counts=counts, noncpu_name_samples=samples)
        return result
    finally:
        value = items = None


def allocator_metadata(torch):
    stats = torch.cuda.memory_stats()
    return dict(device=int(torch.cuda.current_device()), allocated_bytes=int(torch.cuda.memory_allocated()),
        reserved_bytes=int(torch.cuda.memory_reserved()), allocator_backend=torch.cuda.get_allocator_backend(),
        active_bytes=int(stats["active_bytes.all.current"]) if "active_bytes.all.current" in stats else None,
        active_blocks=int(stats["active.all.current"]) if "active.all.current" in stats else None,
        requested_bytes=int(stats["requested_bytes.all.current"]) if "requested_bytes.all.current" in stats else None)


def cleanup_diagnostic(torch, lm, proof, result, check, clear, *, scan=cuda_storage_metadata, collect=gc.collect):
    """Post-runtime observation only: never repair the original cleanup flag."""
    before = q.object_sha(proof)
    result.update(original_owned_cleanup_complete=proof.get("owned_cleanup_complete"),
        documentation="https://docs.pytorch.org/docs/2.10/notes/cuda.html#cublas-workspaces",
        private_api="torch._C._cuda_clearCublasWorkspaces", private_clear_attempts=0,
        admission_open=False, original_proof_overwrite_permitted=False,
        workspace_environment={key: os.environ.get(key) for key in ("CUBLAS_WORKSPACE_CONFIG", "PYTORCH_ALLOC_CONF", "PYTORCH_CUDA_ALLOC_CONF")})
    check()
    result["registered_after_runtime"] = registered_devices(lm.pretrained)
    result["initial_allocator"] = allocator_metadata(torch)
    result["python_cuda_before"] = scan(torch)
    if not result["python_cuda_before"]["complete"]:
        raise q.Hold("bounded Python CUDA metadata incomplete; private clear not attempted")
    def repeat_control():
        check()
        torch.cuda.synchronize()
        collect()
        torch.cuda.empty_cache()
        torch.cuda.synchronize()
        check()
        return allocator_metadata(torch)
    result["after_repeated_control"] = repeat_control()
    result["private_clear_attempts"] = 1
    clear()
    result["private_clear_returned"] = True
    result["after_private_clear_control"] = repeat_control()
    result["python_cuda_after"] = scan(torch)
    result["allocated_delta_control"] = result["initial_allocator"]["allocated_bytes"] - result["after_repeated_control"]["allocated_bytes"]
    result["allocated_delta_private_clear"] = result["after_repeated_control"]["allocated_bytes"] - result["after_private_clear_control"]["allocated_bytes"]
    result["control_allocated_stable"] = result["allocated_delta_control"] == 0
    result["original_proof_unchanged"] = q.object_sha(proof) == before
    if not result["original_proof_unchanged"] or not result["python_cuda_after"]["complete"]:
        raise ValueError("original proof changed or final bounded CUDA metadata incomplete")
    result["diagnostic_complete"] = True
    check()


def pair_metrics(left, right, torch, labels=None):
    """All tensors are detached CPU values; the original right-reference oracle."""
    if (left.device.type != "cpu" or right.device.type != "cpu" or left.dtype != torch.float32
            or right.dtype != torch.float32 or left.ndim != 2 or left.shape != right.shape
            or min(left.shape) <= 0):
        raise ValueError("matching nonempty FP32 CPU matrices required")
    difference = (left - right).abs()
    denominator = q.official.ATOL + q.official.RTOL * right.abs()
    close = torch.isclose(left, right, atol=q.official.ATOL, rtol=q.official.RTOL)
    bad = ~close
    largest = int(difference.argmax())
    index = [largest // left.shape[1], largest % left.shape[1]]
    a, b = left.argmax(dim=1), right.argmax(dim=1)
    result = dict(shape=list(left.shape), original_allclose=bool(torch.allclose(left, right,
        atol=q.official.ATOL, rtol=q.official.RTOL)), atol=q.official.ATOL, rtol=q.official.RTOL,
        normalization="abs(left-right)/(ATOL+RTOL*abs(right)); right is the original torch reference",
        oracle_device="CPU_FP32_SAME_TOLERANCES_NOT_A_REPLACEMENT_FOR_FROZEN_V3_GPU_GATE",
        maximum_absolute_difference=number(difference.max()), maximum_normalized_difference=number((difference / denominator).max()),
        maximum_difference_at=index, maximum_difference_values=[number(left[tuple(index)]), number(right[tuple(index)])],
        violation_count=int(bad.sum()), violations_by_prefix_position=bad.sum(dim=1).tolist(),
        first_violation_indices=bad.nonzero()[:8].tolist(),
        nonfinite_values=[int((~torch.isfinite(value)).sum()) for value in (left, right)],
        argmax_changes=int((a != b).sum()), argmax_changed_positions=(a != b).nonzero().flatten().tolist(),
        left_argmax=a.tolist(), right_argmax=b.tolist())
    if labels is not None:
        if len(labels) != left.shape[0] or len(labels) < 2:
            raise ValueError("same common-prefix next-token labels required")
        target = torch.tensor(labels[1:], dtype=torch.long, device="cpu")
        likelihoods = [-torch.nn.functional.cross_entropy(value[:-1], target, reduction="sum") for value in (left, right)]
        result["same_prefix_label_loglikelihood"] = dict(labels=list(labels[1:]),
            method="FP32_CPU_FULL_VOCAB_CROSS_ENTROPY_SAME_LABELS_NOT_BENCHMARK_SCORE",
            left=number(likelihoods[0]), right=number(likelihoods[1]), difference=number(likelihoods[0] - likelihoods[1]))
    return result


def probe_plan(lm):
    prefix = list(lm.tok_encode("Prefix for a causal state check", add_special_tokens=False))
    tail1 = list(lm.tok_encode(" one future", add_special_tokens=False))
    tail2 = list(lm.tok_encode(" another different future", add_special_tokens=False))
    k = min(len(tail1), len(tail2))
    if (not 2 <= len(prefix) <= 32 or not k or tail1[:k] == tail2[:k]
            or max(len(prefix + tail1), len(prefix + tail2)) > 64):
        raise q.Hold("bounded nonidentical same-length suffixes required; no token remapping")
    return prefix, [("prefix_off", prefix, False), ("prefix_on", prefix, True), ("prefix_repeat", prefix, True),
        ("same_a", prefix + tail1[:k], True), ("same_b", prefix + tail2[:k], True),
        ("same_a_again", prefix + tail1[:k], True), ("original1_off", prefix + tail1, False),
        ("original2_off", prefix + tail2, False), ("original1_on", prefix + tail1, True),
        ("original2_on", prefix + tail2, True)]


def capture(lm, torch, sequence, prefix_size, hidden, check):
    inputs = output = None
    try:
        check()
        inputs = torch.tensor([sequence], dtype=torch.long, device=lm.device)
        with torch.no_grad(), torch.autocast(device_type=lm.device.type, enabled=False):
            output = lm.model(input_ids=inputs, use_cache=False, past_key_values=None, logits_to_keep=0,
                              output_hidden_states=hidden, return_dict=True)
        if (output.past_key_values is not None or output.logits.dtype != torch.float32
                or tuple(output.logits.shape) != (1, len(sequence), lm.model.config.vocab_size)):
            raise ValueError("stateless full FP32 logits required")
        logits = output.logits[0, :prefix_size].detach().cpu().clone()
        states = []
        if hidden:
            if len(output.hidden_states or ()) != lm.model.config.num_hidden_layers + 1:
                raise ValueError("official embedding/layer/final-normalized capture incomplete")
            for value in output.hidden_states:
                if value.dtype != torch.float32 or tuple(value.shape) != (1, len(sequence), lm.model.config.hidden_size):
                    raise ValueError("official hidden capture shape/dtype differs")
                states.append(value[0, :prefix_size].detach().cpu().clone())
        check()
        return dict(logits=logits, hidden=states, input_ids=list(sequence), capture_hidden=hidden)
    finally:
        inputs = output = None


def observe(lm, torch, writer, check):
    prefix, plans = probe_plan(lm)
    shots, rows = {}, []
    for name, sequence, hidden in plans:
        shots[name] = capture(lm, torch, sequence, len(prefix), hidden, check)
        states = shots[name]["hidden"]
        writer.append(dict(kind="snapshot", name=name, input_ids=sequence, prefix_ids=prefix,
            input_shape=[1, len(sequence)], output_hidden_states=hidden,
            hidden_semantics="index0=embedding; intermediate=decoder_output; last=FINAL_NORMALIZED_NOT_PRENORM_LAST_LAYER",
            hidden_values=[dict(index=i, shape=list(value.shape), final_normalized=i == len(states) - 1,
                fp32_sha256=hashlib.sha256(value.contiguous().numpy().tobytes()).hexdigest().upper(),
                first_two_positions_first_eight_features=[[number(v) for v in row] for row in value[:2, :8].tolist()])
                for i, value in enumerate(states)]))
    comparisons = [("capture_flag_toggle", "prefix_off", "prefix_on"),
        ("identical_input_repeat", "prefix_on", "prefix_repeat"),
        ("shape_fixed_future_swap", "same_a", "same_b"), ("shape_fixed_aba", "same_a", "same_a_again"),
        ("shape_changed_original_extension", "prefix_off", "original1_off"),
        ("shape_changed_original_extension", "prefix_off", "original2_off"),
        ("capture_flag_toggle", "original1_off", "original1_on"),
        ("capture_flag_toggle", "original2_off", "original2_on"),
        ("shape_changed_captured_extension", "prefix_on", "original1_on"),
        ("shape_changed_captured_extension", "prefix_on", "original2_on")]
    for category, first, second in comparisons:
        check()
        a, b = shots[first], shots[second]
        row = dict(kind="comparison", category=category, left=first, right=second,
            actual_input_shapes=[[1, len(value["input_ids"])] for value in (a, b)],
            same_input_shape=len(a["input_ids"]) == len(b["input_ids"]), prefix_ids=prefix,
            logits=pair_metrics(a["logits"], b["logits"], torch, prefix), layers=[])
        if a["hidden"] and b["hidden"]:
            if len(a["hidden"]) != len(b["hidden"]):
                raise ValueError("layer capture counts differ")
            row["layers"] = [dict(index=i, final_normalized=i == len(a["hidden"]) - 1,
                feature_argmax_not_token_argmax=True, metrics=pair_metrics(left, right, torch))
                for i, (left, right) in enumerate(zip(a["hidden"], b["hidden"]))]
        writer.append(row)
        rows.append(dict(category=category, left=first, right=second, same_input_shape=row["same_input_shape"],
            original_allclose=row["logits"]["original_allclose"], violation_count=row["logits"]["violation_count"]))
    return dict(snapshots=len(plans), comparisons=rows, numerical_negative_pairs=sum(not row["original_allclose"] for row in rows),
                observations_complete=True, numerical_pass_claim=False, admission_open=False, benchmark_rows=0)


def verify_runtime(proof, assets):
    if any(proof.get(key) is not True for key in ("model_loaded", "body_unchanged", "tokenizer_unchanged", "owned_cleanup_complete")):
        raise ValueError("actual body/tokenizer/owned-cleanup receipts incomplete")
    value = proof.get("key_consumption", {})
    expected = dict(passed=True, consumed_language_keys=320, state_keys=321, tied_head_alias=True,
                    unused_vision_keys=153, unused_mtp_keys=15, unaccounted_keys=0)
    rows, inventory = value.get("language_values", []), assets["inventory"]
    if (any(value.get(key) != item for key, item in expected.items()) or value.get("unused") != inventory["unused"]
            or len(rows) != 320 or {row.get("source") for row in rows} != set(inventory["mapping"])
            or any(row.get("target") != inventory["mapping"][row["source"]]
                or row.get("shape") != inventory["language"][row["source"]]["shape"]
                or row.get("exact_equal") is not True or row.get("loaded_dtype") != "float32"
                or row.get("original_dtype") != inventory["language"][row["source"]]["dtype"]
                or not isinstance(row.get("fp32_value_sha256"), str) or len(row["fp32_value_sha256"]) != 64 for row in rows)):
        raise ValueError("all 320 exact value receipts and 168 explicit-unused headers required")
    token = proof.get("tokenizer_composite_contract", {})
    if (token.get("passed") is not True or token.get("config_added_tokens") != 33
            or token.get("original_json_added_tokens") != 22 or len(token.get("extra_list", [])) != 13
            or len(token.get("named_tokens", {})) != 7):
        raise ValueError("unchanged complete tokenizer proof required")


def recount_observations(directory, manifest, complete):
    counts, seen = dict(snapshot=0, comparison=0), set()
    for shard in manifest["shards"]:
        with (directory / shard["path"]).open(encoding="utf-8") as stream:
            for line in stream:
                row = json.loads(line)
                kind = row["kind"]
                key = (kind, row.get("name"), row.get("left"), row.get("right"))
                if kind not in counts or key in seen:
                    raise ValueError("raw observation kind/uniqueness differs")
                seen.add(key)
                counts[kind] += 1
                if kind == "comparison":
                    shapes, metrics = row["actual_input_shapes"], row["logits"]
                    if (row["same_input_shape"] != (shapes[0] == shapes[1])
                            or sum(metrics["violations_by_prefix_position"]) != metrics["violation_count"]
                            or metrics["original_allclose"] != (metrics["violation_count"] == 0)):
                        raise ValueError("raw shape/original-oracle counts differ")
    if sum(counts.values()) != manifest["rows"] or (complete and counts != dict(snapshot=10, comparison=10)):
        raise ValueError("raw observation inventory incomplete")
    return dict(independent_recount=True, counts=counts)


def finalize(report, directory, signature, pins, admission, started, idle_fn, *,
             identity_fn=identity, verify_fn=q.verify_pins, runtime_fn=verify_runtime,
             clock=time.perf_counter, wall=time.time, timer=q.wall_limit,
             write=q.external.write_json, read=q.read_json, hash_fn=q.digest):
    """Observation-only closure. Never call the frozen Core6 admission finalizer."""
    budget = lambda: q.remaining(admission, started, elapsed=clock() - started, now=wall())
    report_path, closure_path, veto_path = (directory / name for name in ("report.json", "phase_closure.json", "phase_veto.json"))
    original = report.get("original_error")
    try:
        with timer(budget()):
            try:
                report["sources_unchanged"] = identity_fn() == signature
                verify_fn(pins)
                report["assets_and_raw_unchanged"] = True
                if not report["sources_unchanged"]:
                    raise ValueError("source changed before observation closure")
            finally:
                # Preserve actual runtime receipts before any cleanup/source veto.
                report.update(ended_at=datetime.now(timezone.utc).isoformat(), wall_seconds=clock() - started)
                write(report_path, report)
                budget()
                if read(report_path) != q.external.json_ready(report):
                    raise ValueError("stored report differs")
            if report.get("model_load_attempted"):
                runtime_fn(report["model_proof"], report["assets"])
                if idle_fn is None:
                    raise ValueError("NVML cleanup point query missing")
                cleanup = idle_fn()
            else:
                cleanup = None
            if report["returncode"] == 0 and (report.get("observations", {}).get("observations_complete") is not True
                    or report.get("raw_recount") != dict(independent_recount=True, counts=dict(snapshot=10, comparison=10))):
                raise ValueError("complete independently recounted observation inventory missing")
            budget()
            report_sha = hash_fn(report_path)
            closure = dict(schema=SCHEMA + "_CLOSURE", tag=TAG, returncode=report["returncode"], contract=CONTRACT,
                observation_only=True, numerical_pass_claim=False, admission_open=False, source_sha256=signature["sha256"],
                report_sha256=report_sha, owned_cleanup_nvml=cleanup, observed_wall_epoch=wall(),
                observed_elapsed_seconds=clock() - started, atomicity_claim=False)
            write(closure_path, closure)
            if (read(closure_path) != closure or hash_fn(report_path) != report_sha
                    or veto_path.exists() or veto_path.is_symlink()):
                raise ValueError("terminal closure/report changed or veto exists")
            budget()
        budget()
        report.update(phase_closure_verified=True, phase_closure_sha256=hash_fn(closure_path))
        budget()
    except BaseException as error:
        failure = error_snapshot(error)
        report.update(returncode=9, phase_closure_verified=False, closure_error=failure)
        veto = dict(schema=SCHEMA + "_VETO", tag=TAG, returncode=9, source_sha256=signature["sha256"],
            original_error=original, closure_error=failure, contract=CONTRACT, admission_open=False)
        try:
            write(veto_path, veto)
        except BaseException as secondary:
            report["veto_write_error"] = error_snapshot(secondary)

            torch.backends.cudnn.allow_tf32 = False
            writer = ShardWriter(directory, "observations", limit=19_999_999)
            with sdpa_kernel(SDPBackend.MATH), torch.no_grad():
                with q.model_runtime(torch, assets, writer, check, proof, signature) as lm:
                    observations, error = safely(lambda: observe(lm, torch, writer, check))
                    if error is not None:
                        report["original_error"] = error
                    else:
                        report["observations"] = observations
                diagnostic = report["cleanup_diagnostic"] = {}
                if "original_error" not in report:
                    cleanup_diagnostic(torch, lm, proof, diagnostic, check, clear)
                else:
                    diagnostic.update(skipped_after_original_error=True, private_clear_attempts=0, admission_open=False)
            if "original_error" not in report:
                report.update(returncode=0, status="OBSERVATION_COMPLETE_NOT_NUMERICAL_PASS_NO_ADMISSION")
            elif report["original_error"]["type"] == "KeyboardInterrupt":
                report["returncode"] = 130
    except BaseException as error:
        record = error_snapshot(error)
        if "original_error" in report:
            report["secondary_error"] = record
        else:
            report["original_error"] = record
        report["returncode"] = 130 if record["type"] == "KeyboardInterrupt" else 8 if record["type"] == "Hold" else 9
    finally:
        report["model_load_attempted"] = proof.get("model_load_attempted", False)
        report["tokenizer_load_attempted"] = proof.get("tokenizer_load_attempted", False)
        try:
            with q.wall_limit(check()):
                if writer is not None and writer._total_rows:
                    manifest = report["raw_manifest"] = writer.close(lineage=dict(tag=TAG, source_sha256=signature["sha256"]))
                    q.pin_manifest(directory, "observations", manifest, pins)
                    report["raw_recount"] = recount_observations(directory, manifest, report["returncode"] == 0)
        except BaseException as error:
            record = error_snapshot(error)
            report.setdefault("original_error", record)
            report.update(returncode=9, raw_closure_error=record)
        finalize(report, directory, signature, pins, admission, started, idle_fn)
    return {key: report.get(key) for key in ("status", "returncode", "phase_closure_verified", "phase_closure_sha256",
        "observations", "cleanup_diagnostic", "original_error", "closure_error", "contract", "quality_claim", "admission_open", "benchmark_rows")}


def main(argv=None):
    parser = argparse.ArgumentParser(description=__doc__)
    group = parser.add_mutually_exclusive_group()
    group.add_argument("--check-only", action="store_true")
    group.add_argument("--execute", action="store_true")
    parser.add_argument("--source-sha256")
    parser.add_argument("--deadline")
    parser.add_argument("--max-wall-sec", type=int)
    args = parser.parse_args(argv)
    try:
        if not args.execute:
            result = dict(status="SOURCE_ONLY", tag=TAG, source_identity=identity(), models=0, tokenizers=0,
                gpu_calls=0, downloads=0, installs=0, contract=CONTRACT, internal_max_seconds=570)
        else:
            if not args.source_sha256 or not args.deadline or args.max_wall_sec is None:
                raise ValueError("exact source, timezone deadline and whole cap are required")
            result = execute(args.source_sha256, args.deadline, args.max_wall_sec)
        print(json.dumps(q.external.json_ready(result), sort_keys=True))
        return result.get("returncode", 0)
    except BaseException as error:
        code = 8 if isinstance(error, q.Hold) else 9
        print(json.dumps(dict(status="OBSERVER_HOLD" if code == 8 else "OBSERVER_ADMISSION_OR_EXECUTION_ERROR", returncode=code,
            original_error=error_snapshot(error), quality_claim=False, admission_open=False, benchmark_rows=0)))
        return code


if __name__ == "__main__":
    raise SystemExit(main())
