"""INACTIVE ARCHIVE: tests of the rejected manual-index design in /tmp only."""
from pathlib import Path
from contextlib import redirect_stdout
import io
import json
import subprocess
import sys
import tempfile
import unittest
from unittest.mock import patch

sys.path.insert(0, str(Path(__file__).resolve().parent))
import commit_race_record as module


class Tests(unittest.TestCase):
    def setUp(self):
        self.temp = tempfile.TemporaryDirectory(prefix='tl_git_transaction_test_')
        self.addCleanup(self.temp.cleanup)
        self.root = Path(self.temp.name)
        self.base = self.root / 'handoff/audit/nov01_scope_a'
        self.base.mkdir(parents=True)
        self.addCleanup(patch.stopall)
        patch.object(module, 'ROOT', self.root).start()
        patch.object(module, 'BASE', self.base).start()
        self.run_git('init', '-q', '-b', 'main')
        self.run_git('config', 'user.name', 'Fixture')
        self.run_git('config', 'user.email', 'fixture@example.invalid')
        for name in [*module.OWN, 'seed.txt']:
            p = self.root / name
            p.parent.mkdir(parents=True, exist_ok=True)
            p.write_text('original fixture')
        self.run_git('add', '--', *module.OWN, 'seed.txt')
        self.run_git('commit', '-q', '-m', 'fixture baseline')
        self.head = self.run_git('rev-parse', 'HEAD').decode().strip()
        self.index = (self.root / '.git/index').read_bytes()
        for name in module.OWN:
            (self.root / name).write_text('changed fixture')
        (self.base / 'COMMIT_MESSAGE_v3.txt').write_text('fixture closing'+chr(10))
        self.write_record(self.head)

    def run_git(self, *args):
        return subprocess.run(['git', *args], cwd=self.root, check=True,
                              stdout=subprocess.PIPE, stderr=subprocess.PIPE).stdout

    def write_record(self, head):
        (self.base / 'INDEX_RACE_v1.json').write_text(json.dumps(dict(base_head=head)))

    def test_success_publishes_exact_owned_index(self):
        with redirect_stdout(io.StringIO()):
            module.commit()
        self.assertNotEqual(self.run_git('rev-parse', 'HEAD').decode().strip(), self.head)
        self.assertEqual(self.run_git('diff', '--cached', '--name-only'), b'')
        names = {x.decode() for x in self.run_git('diff-tree', '--no-commit-id', '--name-only',
                                                  '--no-renames', '-z', '-r', 'HEAD').split(bytes([0])) if x}
        self.assertEqual(names, set(module.OWN))
        for name in module.OWN:
            self.assertEqual(self.run_git('show', 'HEAD:' + name), (self.root / name).read_bytes())
        self.assertFalse((self.root / '.git/index.lock').exists())

    def test_foreign_staged_work_is_preserved(self):
        (self.root / 'seed.txt').write_text('foreign staged fixture')
        self.run_git('add', '--', 'seed.txt')
        before = (self.root / '.git/index').read_bytes()
        with self.assertRaisesRegex(ValueError, 'nonempty shared index'):
            module.commit()
        self.assertEqual((self.root / '.git/index').read_bytes(), before)
        self.assertEqual(self.run_git('rev-parse', 'HEAD').decode().strip(), self.head)
        self.assertFalse((self.root / '.git/index.lock').exists())

    def test_existing_lock_is_never_removed(self):
        lock = self.root / '.git/index.lock'
        lock.write_bytes(b'foreign fixture lock')
        with self.assertRaises(FileExistsError):
            module.commit()
        self.assertEqual(lock.read_bytes(), b'foreign fixture lock')
        self.assertEqual((self.root / '.git/index').read_bytes(), self.index)

    def test_changed_head_contract_aborts_before_staging(self):
        self.write_record('0' * 40)
        with self.assertRaisesRegex(ValueError, 'HEAD changed'):
            module.commit()
        self.assertEqual((self.root / '.git/index').read_bytes(), self.index)
        self.assertEqual(self.run_git('rev-parse', 'HEAD').decode().strip(), self.head)
        self.assertFalse((self.root / '.git/index.lock').exists())


if __name__ == '__main__':
    unittest.main()
