def commit():
    record = json.loads((BASE / 'INDEX_RACE_v1.json').read_text())
    env = dict(os.environ, GIT_OPTIONAL_LOCKS='0')
    env.pop('GIT_INDEX_FILE', None)
    gitdir = Path(git('rev-parse', '--absolute-git-dir').decode().strip())
    if gitdir != ROOT / '.git' or gitdir.is_symlink():
        raise ValueError('only this exact checkout Git directory is supported')
    index, lock = gitdir / 'index', gitdir / 'index.lock'
    if index.is_symlink() or not index.is_file():
        raise ValueError('regular existing primary index required')
    fd = os.open(lock, os.O_CREAT | os.O_EXCL | os.O_WRONLY | os.O_CLOEXEC, 0o600)
    owned_inode = os.fstat(fd).st_ino
    committed = published = False
    try:
        if git('diff', '--cached', '--name-only', '-z', env=env):
            raise ValueError('nonempty shared index preserved; no staging/commit')
        if git('rev-parse', 'HEAD', env=env).decode().strip() != record['base_head']:
            raise ValueError('HEAD changed after preparation; no commit')
        with tempfile.TemporaryDirectory(prefix='tinylm_scope_a_git_') as temp:
            private_index = Path(temp) / 'index'
            own_env = dict(env, GIT_INDEX_FILE=str(private_index))
            git('read-tree', 'HEAD', env=own_env)
            git('add', '--', *OWN, env=own_env)
            raw = git('diff', '--cached', '--no-renames', '--name-only', '-z', env=own_env)
            names = {part.decode() for part in raw.split(bytes([0])) if part}
            if names != set(OWN):
                raise ValueError('temporary index differs from the explicit paths')
            for name in OWN:
                path = ROOT / name
                if path.is_symlink() or path.stat().st_size >= 20_000_000:
                    raise ValueError('invalid owned payload')
                if git('show', ':' + name, env=own_env) != path.read_bytes():
                    raise ValueError('temporary index bytes changed')
            git('diff', '--cached', '--check', '--', *OWN, env=own_env)
            if git('rev-parse', 'HEAD', env=env).decode().strip() != record['base_head']:
                raise ValueError('HEAD changed before commit')
            git('commit', '--quiet', '-F', str(BASE / 'COMMIT_MESSAGE_v3.txt'), env=own_env)
            committed = True
            # Other normal Git transactions cannot acquire index.lock here.
            # Publish the committed private index using Git's lockfile pattern.
            os.fchmod(fd, stat.S_IMODE(index.stat().st_mode))
            data = private_index.read_bytes()
            with os.fdopen(fd, 'wb') as stream:
                fd = None
                stream.write(data)
                stream.flush()
                os.fsync(stream.fileno())
            if lock.stat().st_ino != owned_inode:
                raise ValueError('primary lock ownership changed')
            os.replace(lock, index)
            published = True
        print(json.dumps(dict(commit=git('rev-parse', 'HEAD').decode().strip(),
                              exact_owned_paths=len(OWN), shared_index_synchronized=True,
                              foreign_staged=0, push_pr=0)))
    finally:

            os.close(fd)
        if not committed and lock.exists() and lock.stat().st_ino == owned_inode:
            lock.unlink()
        if committed and not published:
            print('COMMITTED_BUT_INDEX_SYNC_FAILED: retain owned index.lock for recovery', file=sys.stderr)
