#!/usr/bin/env bash
# ALM fixed-reference student study; not full-vocabulary or truth distillation.
# Fresh source/parent/cache identities and current absolute deadline are enforced.
set -uo pipefail
cd -- "$(dirname -- "${BASH_SOURCE[0]}")"
source scripts/shell/tinylm_env.sh
python_bin="$(tinylm_python)"
export TL_RUNLOG_GIT_SCOPE=codex-safe
stamp="$(date +%H%M%S)"
source_sha="$("$python_bin" -B -X utf8 -c 'from scripts import diag_p106b_alm_train as s; print(s.sha(s.sources()))')"
result=$?
[[ "$result" == 0 && -n "$source_sha" ]] || exit 8
failed=0
for arm in ce alm; do
    echo "[STAGE] Matched2-update cost arm=$arm; fresh parent; checkpoint0."
    timeout --signal=TERM --kill-after=30s 630s "$python_bin" -B -X utf8 scripts/runlog.py --name P106B_Stage7dW_alm_matched_cost_pair_${arm}_"$stamp" --num 101 -- "$python_bin" -B -X utf8 scripts/diag_p106b_alm_train.py --execute --mode cost --arm "$arm" --source-sha256 "$source_sha" --max-wall-sec 600
    result=$?
    if [[ "$result" != 0 ]]; then failed="$result"; fi
done
exit "$failed"
