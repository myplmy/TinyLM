#!/usr/bin/env bash
# Official zero-shot Core6 full eval splits, four preregistered models.
# Requires each closed matching cost report; per-task weighted admission.
# Each model <=2h and hard16:30KST deadline; no training/quality cherry-pick.
set -uo pipefail
cd -- "$(dirname -- "${BASH_SOURCE[0]}")"
source scripts/shell/tinylm_env.sh
python_bin="$(tinylm_python)"
eval_python="data_cache/p106b_official_eval_runtime/core6_env_20261023/bin/python"
export TL_RUNLOG_GIT_SCOPE=codex-safe
stamp="$(date +%H%M%S)"
source_sha="$("$python_bin" -B -X utf8 scripts/diag_p106b_official_core6.py | "$python_bin" -c 'import json,sys;print(json.load(sys.stdin)["source_identity"]["sha256"])')"
result=$?
[[ "$result" == 0 && "$source_sha" =~ ^[A-F0-9]{64}$ ]] || exit 9
failures=0
for model in native595-f native595-q new595-broad-f falcon90m; do
  "$python_bin" -B -X utf8 scripts/runlog.py --name P106B_Stage8eW_core6_full_"$model"_"$stamp" --num 101 -- timeout --signal=TERM --kill-after=30 7250 "$eval_python" -B -X utf8 scripts/diag_p106b_official_core6.py --execute --model "$model" --mode full --source-sha256 "$source_sha"
  result=$?
  if [[ "$result" != 0 ]]; then failures=$((failures+1)); fi
  "$python_bin" -B -X utf8 scripts/runlog.py --name P106B_Stage8eW_core6_full_"$model"_"$stamp" --num 101 --note "Full official evaluation exit=$result; read exact counts, raw LL, truncation and independent audits. Not a BabyLM track or Korean dialogue score."
done
[[ "$failures" == 0 ]]
